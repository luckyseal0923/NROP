# 忘記密碼：跨分頁與返回處理完整修改對照

日期：2026-10-27。新版：NROP-20261027-03（本機完成，本次依使用者要求同步 GitHub；正式站待驗收）。

環境變數已由使用者修正，代理唯讀確認 API_EXTERNAL_URL 與 ADDITIONAL_REDIRECT_URLS 正確。本次修復網頁收到 PKCE code 後只顯示登入的問題，不需 SQL。

原程式只等待 PASSWORD_RECOVERY；目前 SDK 的自動 PKCE 返回處理遺漏 recovery 類型，且原 verifier 僅存於發起的分頁。新程式明確交換 code，以成功交換結果辨識密碼重設，登入憑證維持 sessionStorage，僅短期 verifier 在同來源分頁間共用。

70 項測試與正式版建置通過。包含真實 SDK 搭配合成 HTTP 回應、跨分頁／過期／缺少驗證、成功開框／取消登出及一般登入不誤開框。未實測真實信件至新密碼重新登入，未由代理輸入真實密碼。

部署後確認頁尾為 -03，再重新申請信件，以最新信件在同一瀏覽器點選。舊版只貼回原分頁的建議不足以解決完整流程。

以下提供完整檔案，可直接複製。新增檔案的修改前為不存在。

import { createClient } from '@supabase/supabase-js';

const url = import.meta.env.VITE_SUPABASE_URL;
const key = import.meta.env.VITE_SUPABASE_ANON_KEY;
export const supabase = url && key ? createClient(url, key, {
  auth: { storage: window.sessionStorage, persistSession: true, autoRefreshToken: true, flowType: 'pkce' },
  global: { fetch: (input,init={})=>fetch(input,{...init,signal:init.signal||AbortSignal.timeout(15000)}) }
}) : null;

export async function rpc(name, params = {}) {
  if (!supabase) throw new Error('尚未設定 Supabase，請參閱安裝說明設定 .env.local。');
  const { data, error } = await supabase.rpc(name, params);
  if (error) throw error;
  return data;
}

// 獨立、僅存記憶體的驗證工作階段，避免切換主畫面正在使用的帳號。
export async function confirmBedChange({requestId,email,password,expectedUserId}) {
  if(!supabase)throw new Error('尚未設定 Supabase');
  const confirmation=createClient(url,key,{
    auth:{persistSession:false,autoRefreshToken:false,detectSessionInUrl:false,storageKey:'nrop-bed-confirm-'+crypto.randomUUID()},
    global:{fetch:(input,init={})=>fetch(input,{...init,signal:init.signal||AbortSignal.timeout(15000)})}
  });
  try {
    const {data,error}=await confirmation.auth.signInWithPassword({email,password});
    if(error)throw error;
    if(data.user?.id!==expectedUserId)throw new Error('請輸入目前登入系統管理者本人的帳號密碼');
    const result=await confirmation.rpc('nrop_confirm_bed_change',{p_request:requestId});
    if(result.error)throw result.error;
    return result.data;
  } finally {
    // 只登出此次確認用的工作階段；不登出其他分頁或平台。
    try{await confirmation.auth.signOut({scope:'local'});}catch{}
  }
}

// 申請前先確認 Auth 已取消驗證信，避免仍沿用舊設定而寄出使用者不要的信件。
export async function requireApprovalOnlySignup() {
  const res=await fetch(url.replace(/\/$/,'')+'/auth/v1/settings',{headers:{apikey:key},signal:AbortSignal.timeout(15000)});
  if(!res.ok)throw new Error('無法確認註冊設定，請稍後再試。');
  const settings=await res.json();
  if(settings.mailer_autoconfirm!==true)throw new Error('管理者審核註冊尚未啟用，請管理者將 Supabase 的 ENABLE_EMAIL_AUTOCONFIRM 設為 true 並重啟後再申請。');
}

export function errorMessage(error) {
  const message = error?.message || '連線失敗，請稍後再試。';
  if (/Invalid login credentials/i.test(message)) return 'Email 或密碼錯誤。';
  if (/Email not confirmed/i.test(message)) return '此帳號的註冊尚未完成，請在管理者啟用免驗證信設定後，使用原 Email 重新申請。';
  if (/User already registered/i.test(message)) return '此 Email 已有帳號，請直接登入。';
  if (/Failed to fetch|NetworkError|fetch failed|503|Site not available|Unexpected token|timed out|aborted/i.test(message)) return 'Supabase 暫時無法連線，請檢查服務狀態後重試。';
  if (/Could not find the function|schema cache/i.test(message)) return '尚未安裝 NROP 資料庫，請先在 SQL Editor 執行 INSTALL.sql。';
  if (/rate limit/i.test(message)) return '嘗試次數過多，請稍後再試。';
  if (/signups not allowed/i.test(message)) return 'Supabase 尚未開放註冊，請聯絡系統管理者。';
  return message.length > 240 ? '操作失敗，請檢查服務狀態或聯絡系統管理者。' : message;
}

// 登入憑證仍限本分頁；只有短期 PKCE 驗證資料供同來源的信件分頁使用。
export function createAuthStorage({session,shared,storageKey,now=Date.now,ttl=60*60*1000}) {
 const verifierKey=storageKey+'-code-verifier',sharedKey='nrop-pkce:'+verifierKey;
 const removeShared=()=>{try{shared().removeItem(sharedKey);}catch{}};
 return {
  getItem(key){
   if(key!==verifierKey)return session.getItem(key);
   let raw=null;
   try{raw=shared().getItem(sharedKey);}catch{}
   if(raw!==null){
    let item;
    try{item=JSON.parse(raw);}catch{}
    if(typeof item?.value==='string'&&Number.isFinite(item.expiresAt)&&item.expiresAt>now())return item.value;
    removeShared();session.removeItem(key);return null;
   }
   // 相容升級前已在原分頁發起、尚未完成的流程。
   return session.getItem(key);
  },
  setItem(key,value){
   if(key!==verifierKey){session.setItem(key,value);return;}
   session.removeItem(key);removeShared();
   try{shared().setItem(sharedKey,JSON.stringify({value,expiresAt:now()+ttl}));}
   catch{session.setItem(key,value);}
  },
  removeItem(key){
   session.removeItem(key);
   if(key===verifierKey)removeShared();
  }
 };
}

// 明確交換 PKCE code，使用交換結果的 recovery 類型，不依賴 SDK 初始化事件。
export async function completeAuthRedirect({auth,location,history}) {
 const url=new URL(location.href),hash=new URLSearchParams(url.hash.slice(1));
 const code=url.searchParams.get('code');
 const hasError=url.searchParams.has('error')||url.searchParams.has('error_description')||hash.has('error')||hash.has('error_description');
 if(!code&&!hasError)return {recovery:false};
 try{
  if(hasError)throw new Error('驗證連結已失效，請重新按「忘記密碼」取得最新信件。');
  const {data,error}=await auth.exchangeCodeForSession(code);
  if(error){
   if(error.name==='AuthPKCECodeVerifierMissingError'||error.code==='pkce_code_verifier_not_found')
    throw new Error('找不到這次重設的驗證資料，請用原本的瀏覽器重新按「忘記密碼」，再開啟最新信件。若瀏覽器封鎖儲存空間，請在原本的 NROP 分頁開啟連結。');
   throw new Error('驗證連結無法完成驗證，可能已過期、已使用或有較新的申請。請重新按「忘記密碼」取得最新信件。');
  }
  if(!data?.session)throw new Error('未取得有效的驗證工作階段，請重新按「忘記密碼」。');
  return {recovery:data.redirectType==='PASSWORD_RECOVERY'};
 }finally{
  for(const key of ['code','error','error_code','error_description'])url.searchParams.delete(key);
  if(hasError){
   for(const key of ['error','error_code','error_description'])hash.delete(key);
   url.hash=hash.toString();
  }
  history.replaceState(history.state,'',url.toString());
 }
}
